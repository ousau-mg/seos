import React, { useEffect, useRef, useState } from 'react';

export function TrilhasEmbed() {
  const containerRef = useRef(null);
  const iframeRef = useRef(null);
  const [aceitouAviso, setAceitouAviso] = useState(false);

  useEffect(() => {
    // Verifica se o usuário já aceitou o aviso nesta sessão (seguro para SSR/Docusaurus)
    if (sessionStorage.getItem('seos_trilhas_aviso') === 'true') {
      setAceitouAviso(true);
    }

    const ajustarEscala = () => {
      if (!containerRef.current || !iframeRef.current) return;
      const larguraDisponivel = containerRef.current.offsetWidth;

      if (window.innerWidth <= 768) {
        const escala = larguraDisponivel / 1200;
        iframeRef.current.style.transform = `scale(${escala})`;
        iframeRef.current.style.transformOrigin = 'top left';
        containerRef.current.style.height = `${3000 * escala}px`;
      } else {
        iframeRef.current.style.transform = 'none';
        containerRef.current.style.height = '1400px';
      }
    };

    ajustarEscala();
    window.addEventListener('resize', ajustarEscala);
    return () => window.removeEventListener('resize', ajustarEscala);
  }, []);

  const handleAceitar = () => {
    setAceitouAviso(true);
    sessionStorage.setItem('seos_trilhas_aviso', 'true');
  };

  return (
    <div>
      {/* Faixa discreta que permanece no topo mesmo após fechar o overlay */}
      <div className="trilhas-banner-status">
        <strong>Versão preliminar:</strong> Este mapeamento de trilhas está em desenvolvimento e validação pela equipe SEOS/MG.
      </div>

      <div ref={containerRef} className="trilhas-iframe-container">
        {!aceitouAviso && (
          <div className="trilhas-overlay">
            <div className="trilhas-overlay-card">
              <span className="trilhas-badge">Em implantação</span>
              <h3>Mapeamento Preliminar das Trilhas</h3>
              <p>
                Esta seção reúne a estrutura das trilhas de capacitação do SEOS/MG. Alguns módulos ainda estão em fase de produção ou revisão e serão disponibilizados progressivamente.
              </p>
              <button
                type="button"
                className="button button--primary"
                onClick={handleAceitar}
              >
                Entendi, visualizar trilhas
              </button>
            </div>
          </div>
        )}

        <iframe
          ref={iframeRef}
          src="/seos/trilhas.html"
          title="Trilhas de Capacitação"
        />
      </div>
    </div>
  );
}
