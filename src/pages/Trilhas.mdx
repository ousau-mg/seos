import React, { useEffect, useRef, useState } from 'react';
import useBaseUrl from '@docusaurus/useBaseUrl';
import StatusNotice from '@site/src/components/StatusNotice';

export function TrilhasEmbed() {
  const containerRef = useRef(null);
  const iframeRef = useRef(null);
  const homeUrl = useBaseUrl('/');
  const [mostrarOverlay, setMostrarOverlay] = useState(false);

  useEffect(() => {
    if (sessionStorage.getItem('seos_trilhas_aviso') !== 'true') {
      setMostrarOverlay(true);
    }
  }, []);

  useEffect(() => {
    const ajustarEscala = () => {
      if (!containerRef.current || !iframeRef.current) return;
      const larguraDisponivel = containerRef.current.offsetWidth;

      if (window.innerWidth <= 768) {
        const escala = larguraDisponivel / 1200;
        iframeRef.current.style.transform = `scale(${escala})`;
        iframeRef.current.style.transformOrigin = 'top left';
        containerRef.current.style.height = `${3000 * escala}px`;
      } else {
        iframeRef.current.style.transform = 'none';
        containerRef.current.style.height = '1400px';
      }
    };

    ajustarEscala();
    window.addEventListener('resize', ajustarEscala);
    return () => window.removeEventListener('resize', ajustarEscala);
  }, [mostrarOverlay]);

  const liberarPrevia = () => {
    setMostrarOverlay(false);
    sessionStorage.setItem('seos_trilhas_aviso', 'true');
  };

  return (
    <div>
      {mostrarOverlay ? (
        /* ESTADO 1 (IF): Apenas o overlay centralizado enquanto o usuário não clicou */
        <div ref={containerRef} className="trilhas-iframe-container">
          <StatusNotice
            overlay={true}
            eyebrow="SEOS/MG · Prévia"
            badge="Em implantação"
            title="Trilhas de Capacitação em Implantação"
            description="Esta página exibe a estrutura inicial das trilhas para o SUS/MG. Alguns conteúdos ainda estão em produção ou revisão."
            buttonText="Visualizar trilhas"
            onConfirm={liberarPrevia}
            secondaryText="Voltar ao início"
            secondaryHref={homeUrl}
          />

          <iframe
            ref={iframeRef}
            src="/seos/trilhas.html"
            title="Trilhas de Capacitação"
            style={{ pointerEvents: 'none' }}
          />
        </div>
      ) : (
        /* ESTADO 2 (ELSE): Após clicar, mostra o aviso discreto no topo + o mapa livre */
        <div>
          <StatusNotice
            eyebrow="Trilhas de Capacitação"
            badge="Em implantação"
            title="Mapeamento preliminar em validação"
            description="Parte dos materiais já está disponível e outros módulos estão em produção ou atualização pela equipe SEOS/MG."
            compact={true}
          />

          <div ref={containerRef} className="trilhas-iframe-container">
            <iframe
              ref={iframeRef}
              src="/seos/trilhas.html"
              title="Trilhas de Capacitação"
              style={{ pointerEvents: 'auto' }}
            />
          </div>
        </div>
      )}
    </div>
  );
}

<TrilhasEmbed />
