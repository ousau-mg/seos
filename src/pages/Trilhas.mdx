import React, { useEffect, useRef } from 'react';

export function TrilhasEmbed() {
  const containerRef = useRef(null);
  const iframeRef = useRef(null);

  useEffect(() => {
    const ajustarEscala = () => {
      if (!containerRef.current || !iframeRef.current) return;
      const larguraDisponivel = containerRef.current.offsetWidth;

      if (window.innerWidth <= 768) {
        const escala = larguraDisponivel / 1200;
        iframeRef.current.style.transform = `scale(${escala})`;
        iframeRef.current.style.transformOrigin = 'top left';
        containerRef.current.style.height = `${3000 * escala}px`;
      } else {
        iframeRef.current.style.transform = 'none';
        containerRef.current.style.height = '1400px';
      }
    };

    ajustarEscala();
    window.addEventListener('resize', ajustarEscala);
    return () => window.removeEventListener('resize', ajustarEscala);
  }, []);

  return (
    <div ref={containerRef} className="trilhas-iframe-container">
      <iframe
        ref={iframeRef}
        src="/seos/trilhas.html"
        title="Trilhas de Capacitação"
      />
    </div>
  );
}
